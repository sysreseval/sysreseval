 30class TranslatedText(dict):
 31    """A dict subclass holding translations keyed by language code (e.g. 'en', 'fr').
 32
 33    Use ``resolve(lang)`` to get the text for a specific language, falling back to the
 34    first available translation.  ``from_value`` ensures backward-compat with plain strings.
 35    """
 36
 37    def resolve(self, lang: str, fallback: str = '') -> str:
 38        return self.get(lang) or next(iter(self.values()), fallback)
 39
 40    def resolve_priority(self, langs: list, fallback: str = '') -> str:
 41        """Return the first translation found in the priority list, or first available."""
 42        for lang in langs:
 43            if self.get(lang):
 44                return self[lang]
 45        return next(iter(self.values()), fallback)
 46
 47    def __add__(self, other) -> 'TranslatedText':
 48        if isinstance(other, str):
 49            return TranslatedText({lang: self[lang] + other for lang in self})
 50        if not isinstance(other, TranslatedText):
 51            return NotImplemented
 52        if not self:
 53            return TranslatedText(other)
 54        if not other:
 55            return TranslatedText(self)
 56        # Merge mismatched language sets gracefully (partial translations): take
 57        # the union of languages, falling back to each side's default-language
 58        # (first-key) value for a language that side lacks.
 59        self_default = next(iter(self.values()))
 60        other_default = next(iter(other.values()))
 61        langs = list(dict.fromkeys((*self, *other)))
 62        return TranslatedText({
 63            lang: self.get(lang, self_default) + other.get(lang, other_default)
 64            for lang in langs
 65        })
 66
 67    def __radd__(self, other) -> 'TranslatedText':
 68        if isinstance(other, str):
 69            return TranslatedText({lang: other + self[lang] for lang in self})
 70        return NotImplemented
 71
 72    def format(self, **kwargs) -> 'TranslatedText':
 73        """Apply str.format(**kwargs) to every language value."""
 74        return TranslatedText({lang: text.format(**kwargs) for lang, text in self.items()})
 75
 76    @classmethod
 77    def from_value(cls, v, default_lang: str = 'en') -> 'TranslatedText':
 78        if isinstance(v, cls):
 79            return v
 80        if isinstance(v, dict):
 81            return cls(v)
 82        return cls({default_lang: str(v)}) if v else cls()

217def form_field_lines(spec: str) -> int | None:
218    """Visible lines of a multi-line form field spec (``+`` or ``+N``), None for any other spec.
219
220    ``+`` and ``+0`` give ``params.default_form_field_lines``; what follows the digits is ignored.

226    return int(m.group(1) or 0) or params.default_form_field_lines

290def bonus_texts(label: str, max_text: str, bonus: bool, in_prefix: bool = False,
291                word: str = "Bonus") -> tuple[str, str]:
292    """The ``(label, max)`` texts of the row of a grade element in a report.
293
294    Unchanged unless *bonus*: then the maximum reads ``"2 (Bonus)"`` (``"(Bonus)"`` when
295    *max_text* is empty, e.g. in letter mode), or, with *in_prefix*, the label reads
296    ``"Bonus: label"`` and the maximum is left untouched (``--bonus-in-prefix`` of
297    ``sre outline`` / ``sre sheet`` / ``sre watch``).  *word* is the already translated
298    "Bonus" of the caller (gettext, ``QObject.tr``).
299    """
300    if not bonus:
301        return label, max_text
302    if in_prefix:
303        return f"{word}: {label}", max_text
304    return label, f"{max_text} ({word})".strip()
