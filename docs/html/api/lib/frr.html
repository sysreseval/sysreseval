lib.frr

FRR / OSPF / BGP grading helpers.

Every get_* function registers one command through grade.test() -- so it follows the two-pass contract of Grade.grade() (placeholder result on the registration pass, real output afterwards) -- and hands the text to a pure parse_* function. The parsers are unit-tested on captured FRR 9 output in tests/test_frr.py. The OSPF helpers parse the text output of vtysh, the BGP helpers its json output (show bgp ipv4 unicast ... json). On the placeholder output every parser returns its empty shape ({}/[]), never an exception.

Pass allow_error=True when the router may not run OSPF/BGP yet (vtysh then exits non-zero and the error would otherwise be recorded in the archive).

  59def parse_ospf_interfaces(output: str) -> Dict[str, Dict[str, Any]]:
  60    """Parse the text of ``show ip ospf interface`` (see :func:`get_ospf_interfaces`)."""
  61    result: Dict[str, Dict[str, Any]] = {}
  62    current: Dict[str, Any] | None = None
  63
  64    for line in output.splitlines():
  65        # Interface header: "ethX is up"
  66        m = re.match(r'^(\S+) is (up|down)', line)
  67        if m:
  68            current = {
  69                'state': m.group(2),
  70                'ospf_enabled': True,
  71                'mtu': None, 'bandwidth': None,
  72                'internet_address': None, 'broadcast': None, 'area': None,
  73                'router_id': None, 'network_type': None, 'cost': None,
  74                'ospf_state': None, 'priority': None, 'passive': False,
  75                'hello_interval': None, 'dead_interval': None, 'retransmit_interval': None,
  76                'neighbor_count': None, 'adjacent_neighbor_count': None,
  77                'dr_id': None, 'dr_address': None,
  78                'bdr_id': None, 'bdr_address': None,
  79            }
  80            result[m.group(1)] = current
  81            continue
  82
  83        if current is None:
  84            continue
  85
  86        if 'OSPF not enabled on this interface' in line:
  87            current['ospf_enabled'] = False
  88            continue
  89
  90        # ifindex 13, MTU 1500 bytes, BW 10000 Mbit <UP,LOWER_UP,BROADCAST,RUNNING,MULTICAST>
  91        m = re.search(r'MTU (\d+) bytes,\s*BW (\d+) Mbit', line)
  92        if m:
  93            current['mtu'] = int(m.group(1))
  94            current['bandwidth'] = int(m.group(2))
  95            continue
  96
  97        # Internet Address 10.150.180.224/28, Broadcast 10.150.180.239, Area 0.0.0.0
  98        m = re.search(r'Internet Address (\S+?),\s*Broadcast (\S+?),\s*Area (\S+)', line)
  99        if m:
 100            current['internet_address'] = m.group(1)
 101            current['broadcast'] = m.group(2)
 102            current['area'] = m.group(3)
 103            continue
 104
 105        # Internet Address 10.0.0.1/30, Area 0.0.0.0   (point-to-point, no broadcast)
 106        m = re.search(r'Internet Address (\S+?),\s*Area (\S+)', line)
 107        if m:
 108            current['internet_address'] = m.group(1)
 109            current['area'] = m.group(2)
 110            continue
 111
 112        # Router ID 10.150.180.224, Network Type BROADCAST, Cost: 10
 113        m = re.search(r'Router ID (\S+?),\s*Network Type (\S+?),\s*Cost:\s*(\d+)', line)
 114        if m:
 115            current['router_id'] = m.group(1)
 116            current['network_type'] = m.group(2)
 117            current['cost'] = int(m.group(3))
 118            continue
 119
 120        # Transmit Delay is 1 sec, State Backup, Priority 1
 121        m = re.search(r'State (\S+?),\s*Priority (\d+)', line)
 122        if m:
 123            current['ospf_state'] = m.group(1).rstrip(',')
 124            current['priority'] = int(m.group(2))
 125            continue
 126
 127        # Designated Router (ID) 172.23.52.228 Interface Address 10.150.180.230/28
 128        m = re.search(r'Designated Router \(ID\) (\S+)\s+Interface Address (\S+)', line)
 129        if m and 'Backup' not in line:
 130            current['dr_id'] = m.group(1)
 131            current['dr_address'] = m.group(2)
 132            continue
 133
 134        # Backup Designated Router (ID) 10.150.180.224, Interface Address 10.150.180.224
 135        m = re.search(r'Backup Designated Router \(ID\) (\S+?),\s*Interface Address (\S+)', line)
 136        if m:
 137            current['bdr_id'] = m.group(1)
 138            current['bdr_address'] = m.group(2)
 139            continue
 140
 141        # Timer intervals configured, Hello 10s, Dead 40s, Wait 40s, Retransmit 5
 142        m = re.search(r'Hello (\d+)s,\s*Dead (\d+)s,.*Retransmit (\d+)', line)
 143        if m:
 144            current['hello_interval'] = int(m.group(1))
 145            current['dead_interval'] = int(m.group(2))
 146            current['retransmit_interval'] = int(m.group(3))
 147            continue
 148
 149        # No Hellos (Passive interface)
 150        if 'No Hellos (Passive interface)' in line:
 151            current['passive'] = True
 152            continue
 153
 154        # Neighbor Count is 1, Adjacent neighbor count is 1
 155        m = re.search(r'Neighbor Count is (\d+),\s*Adjacent neighbor count is (\d+)', line)
 156        if m:
 157            current['neighbor_count'] = int(m.group(1))
 158            current['adjacent_neighbor_count'] = int(m.group(2))
 159            continue
 160
 161    return result

 204def parse_ospf_neighbors(output: str) -> List[Dict[str, Any]]:
 205    """Parse the text of ``show ip ospf neighbor``.
 206
 207    Data rows look like (the "Up Time" column exists in FRR >= 7.5 only)::
 208
 209        Neighbor ID     Pri State           Up Time         Dead Time Address         Interface                        RXmtL RqstL DBsmL
 210        10.0.0.2          1 Full/DR         2m34s             36.245s 10.0.12.2       eth0:10.0.12.1                       0     0     0
 211
 212    Rows are parsed from the end (three counters, interface, address, dead time) so
 213    both layouts work.  Non-data lines are skipped.
 214    """
 215    result: List[Dict[str, Any]] = []
 216    for line in output.splitlines():
 217        fields = line.split()
 218        if len(fields) < 9 or not _IPV4_RE.match(fields[0]):
 219            continue
 220        state, _, role = fields[2].partition('/')
 221        interface, _, local_address = fields[-4].partition(':')
 222        try:
 223            priority: Optional[int] = int(fields[1])
 224        except ValueError:
 225            priority = None
 226        result.append({
 227            'neighbor_id': fields[0],
 228            'priority': priority,
 229            'state': state,
 230            'role': role or None,
 231            'up_time': fields[3] if len(fields) >= 10 else None,
 232            'dead_time': fields[-6],
 233            'address': fields[-5],
 234            'interface': interface,
 235            'local_address': local_address or None,
 236        })
 237    return result

 252def find_neighbor(neighbors: List[Dict[str, Any]], neighbor_id: str = None, interface: str = None,
 253                  address: str = None) -> Optional[Dict[str, Any]]:
 254    """Return the first neighbour of :func:`get_ospf_neighbors` matching every given criterion."""

 277def parse_ospf_info(output: str) -> Dict[str, Any]:
 278    """Parse the text of ``show ip ospf``.
 279
 280    Returns ``{}`` when no router id is found (OSPF not running), else::
 281
 282        {'router_id': '10.0.0.4', 'abr': True, 'asbr': False,
 283         'areas': {'0.0.0.0': {'type': 'backbone', 'no_summary': False, 'interfaces': 3,
 284                               'active_interfaces': 3, 'full_neighbors': 2,
 285                               'authentication': 'no'}, ...}}
 286
 287    ``type`` is ``'backbone'``, ``'stub'``, ``'nssa'`` or ``'normal'``;
 288    ``authentication`` is ``'no'``, ``'simple password'`` or ``'message digest'``.
 289    """
 290    info: Dict[str, Any] = {'router_id': None, 'abr': False, 'asbr': False, 'areas': {}}
 291    current: Dict[str, Any] | None = None
 292
 293    for line in output.splitlines():
 294        m = re.search(r'OSPF Routing Process, Router ID:\s*(\S+)', line)
 295        if m:
 296            info['router_id'] = m.group(1)
 297            continue
 298        if 'This router is an ABR' in line:
 299            info['abr'] = True
 300            continue
 301        if 'This router is an ASBR' in line:
 302            info['asbr'] = True
 303            continue
 304        m = re.match(r'^\s*Area ID:\s*(\S+)(?:\s*\((.*?)\))?', line)
 305        if m:
 306            desc = (m.group(2) or '').strip()
 307            low = desc.lower()
 308            if low.startswith('backbone'):
 309                area_type = 'backbone'
 310            elif low.startswith('stub'):
 311                area_type = 'stub'
 312            elif low.startswith('nssa'):
 313                area_type = 'nssa'
 314            else:
 315                area_type = 'normal'
 316            current = {
 317                'type': area_type,
 318                'no_summary': 'no summary' in low,
 319                'interfaces': None, 'active_interfaces': None,
 320                'full_neighbors': None, 'authentication': None,
 321            }
 322            info['areas'][m.group(1)] = current
 323            continue
 324        if current is None:
 325            continue
 326        m = re.search(r'Number of interfaces in this area:\s*Total:\s*(\d+),\s*Active:\s*(\d+)', line)
 327        if m:
 328            current['interfaces'] = int(m.group(1))
 329            current['active_interfaces'] = int(m.group(2))
 330            continue
 331        m = re.search(r'Number of fully adjacent neighbors in this area:\s*(\d+)', line)
 332        if m:
 333            current['full_neighbors'] = int(m.group(1))
 334            continue
 335        m = re.search(r'Area has (.+?) authentication', line)
 336        if m:
 337            current['authentication'] = m.group(1).strip()
 338            continue
 339
 340    if info['router_id'] is None:
 341        return {}
 342    return info

 368# "N IA 10.0.5.0/24 [30] area: 0.0.0.0", "D IA 10.37.4.0/22 Discard entry",
 369# "N E2 203.0.113.0/24 [10/20] tag: 0" and, in the router table, "R 1.1.1.1 IA [400] area: 0.0.0.2, ASBR"
 370# (the type marker follows the router id there).
 371_OSPF_ROUTE_RE = re.compile(
 372    r'^(?P<kind>[NRD])\s+(?P<type>IA|E1|E2|N1|N2)?\s*'
 373    r'(?P<dest>\d{1,3}(?:\.\d{1,3}){3}(?:/\d{1,2})?)\s+'
 374    r'(?:(?P<type2>IA|E1|E2|N1|N2)\s+)?'
 375    r'(?:\[(?P<cost>\d+)(?:/(?P<ext_cost>\d+))?\]|(?P<discard>Discard entry))'
 376    r'(?P<rest>.*)$'
 377)

 380def parse_ospf_routes(output: str) -> Dict[str, Dict[str, Any]]:
 381    """Parse the text of ``show ip ospf route``.
 382
 383    Network entries (``N``) and discard entries (``D``, created by ``area range``) are
 384    keyed by prefix, router entries (``R``) by router id::
 385
 386        {'10.0.5.0/24': {'kind': 'N', 'type': 'IA', 'cost': 30, 'ext_cost': None, 'tag': None,
 387                         'area': '0.0.0.0', 'abr': False, 'asbr': False,
 388                         'nexthops': [('10.0.12.3', 'eth0')]},
 389         '203.0.113.0/24': {'kind': 'N', 'type': 'E2', 'cost': 10, 'ext_cost': 20, 'tag': 0, ...},
 390         '10.0.0.3': {'kind': 'R', 'type': 'intra', 'cost': 10, 'area': '0.0.0.0', 'abr': True, ...}}
 391
 392    ``type`` is ``'intra'`` for plain intra-area entries, ``'IA'``, ``'E1'``, ``'E2'``,
 393    ``'N1'`` or ``'N2'``.  ``nexthops`` holds ``(ip, interface)`` pairs, ``ip`` being
 394    ``None`` for ``directly attached to`` entries.
 395    """
 396    result: Dict[str, Dict[str, Any]] = {}
 397    current: Dict[str, Any] | None = None
 398
 399    for line in output.splitlines():
 400        m = _OSPF_ROUTE_RE.match(line.strip())
 401        if m:
 402            rest = m.group('rest') or ''
 403            area_m = re.search(r'area:\s*(\S+?)(?:,|\s|$)', rest)
 404            tag_m = re.search(r'tag:\s*(\d+)', rest)
 405            current = {
 406                'kind': m.group('kind'),
 407                'type': m.group('type') or m.group('type2') or 'intra',
 408                'cost': int(m.group('cost')) if m.group('cost') else None,
 409                'ext_cost': int(m.group('ext_cost')) if m.group('ext_cost') else None,
 410                'tag': int(tag_m.group(1)) if tag_m else None,
 411                'area': area_m.group(1) if area_m else None,
 412                'abr': 'ABR' in rest,
 413                'asbr': 'ASBR' in rest,
 414                'discard': bool(m.group('discard')),
 415                'nexthops': [],
 416            }
 417            result[m.group('dest')] = current
 418            continue
 419        if current is None:
 420            continue
 421        m = re.search(r'via (\S+?),\s*(\S+)', line)
 422        if m:
 423            current['nexthops'].append((m.group(1), m.group(2)))
 424            continue
 425        m = re.search(r'directly attached to (\S+)', line)
 426        if m:
 427            current['nexthops'].append((None, m.group(1)))
 428            continue
 429
 430    return result

 446def parse_kernel_routes(output: str) -> Dict[str, Dict[str, Any]]:
 447    """Parse the JSON of ``ip -j route``.
 448
 449    Returns ``{prefix: {'protocol': str, 'metric': int, 'scope': str | None,
 450    'nexthops': [(gateway | None, dev)]}}``; ``default`` becomes ``0.0.0.0/0`` and
 451    host routes get ``/32``.  ECMP routes list every next hop.  When the same prefix
 452    appears several times (different metrics) the first entry wins.
 453    """
 454    try:
 455        routes = json.loads(output)
 456    except (ValueError, TypeError):
 457        return {}
 458    if not isinstance(routes, list):
 459        return {}
 460    result: Dict[str, Dict[str, Any]] = {}
 461    for r in routes:
 462        if not isinstance(r, dict):
 463            continue
 464        dst = r.get('dst')
 465        if not dst:
 466            continue
 467        if dst == 'default':
 468            dst = '0.0.0.0/0'
 469        elif '/' not in dst:
 470            dst += '/32'
 471        if dst in result:
 472            continue
 473        if isinstance(r.get('nexthops'), list):
 474            nexthops = [(n.get('gateway'), n.get('dev')) for n in r['nexthops'] if isinstance(n, dict)]
 475        else:
 476            nexthops = [(r.get('gateway'), r.get('dev'))]
 477        result[dst] = {
 478            'protocol': str(r.get('protocol', '')),
 479            'metric': int(r.get('metric', 0) or 0),
 480            'scope': r.get('scope'),
 481            'nexthops': nexthops,
 482        }
 483    return result

 494def kernel_route_gateways(routes: Dict[str, Dict[str, Any]], prefix, ospf_only: bool = True,
 495                          protocols=None) -> List[str]:
 496    """Return the gateway list of ``prefix`` in a :func:`parse_kernel_routes` result.
 497
 498    ``[]`` when the prefix is absent or when its protocol is not accepted: by default
 499    only OSPF routes are (``ospf_only``); ``protocols`` (e.g. ``BGP_ROUTE_PROTOCOLS``)
 500    replaces that filter, ``ospf_only=False`` and ``protocols=None`` accept any route.
 501    """
 502    entry = routes.get(str(prefix))
 503    if entry is None:
 504        return []
 505    if protocols is not None:
 506        if str(entry['protocol']) not in protocols:
 507            return []
 508    elif ospf_only and not is_ospf_protocol(entry['protocol']):
 509        return []
 510    return [gw for gw, _dev in entry['nexthops'] if gw]

 551def _parse_router_bgp_line(rb: Dict[str, Any], words: List[str], neighbor_cfg) -> None:
 552    """Apply one indented ``router bgp`` statement (router level or address-family) to ``rb``."""
 553    if not words:
 554        return
 555    negated = words[0] == 'no'
 556    w = words[1:] if negated else words
 557    if not w:
 558        return
 559    if w[0] in ('address-family', 'exit-address-family'):
 560        return
 561    if w[0] == 'neighbor' and len(w) >= 3:
 562        n = neighbor_cfg(w[1])
 563        kw, arg = w[2], (w[3] if len(w) >= 4 else None)
 564        if kw == 'remote-as' and arg:
 565            n['remote_as'] = int(arg) if arg.isdigit() else arg
 566        elif kw == 'peer-group' and arg:
 567            n['peer_group'] = arg
 568        elif kw == 'description':
 569            n['description'] = " ".join(w[3:]) or None
 570        elif kw == 'update-source' and arg:
 571            n['update_source'] = arg
 572        elif kw == 'timers' and len(w) >= 5 and w[3].isdigit() and w[4].isdigit():
 573            n['timers'] = (int(w[3]), int(w[4]))
 574        elif kw == 'shutdown':
 575            n['shutdown'] = not negated
 576        elif kw == 'password':
 577            n['password'] = None if negated else arg
 578        elif kw == 'ebgp-multihop':
 579            n['ebgp_multihop'] = None if negated else (int(arg) if arg and arg.isdigit() else 255)
 580        elif kw == 'next-hop-self':
 581            n['next_hop_self'] = not negated
 582        elif kw == 'route-reflector-client':
 583            n['route_reflector_client'] = not negated
 584        elif kw == 'default-originate':
 585            n['default_originate'] = not negated
 586            if not negated and len(w) >= 5 and w[3] == 'route-map':
 587                n['default_originate_route_map'] = w[4]
 588        elif kw == 'route-map' and len(w) >= 5 and w[4] in ('in', 'out'):
 589            n[f'route_map_{w[4]}'] = None if negated else w[3]
 590        elif kw == 'prefix-list' and len(w) >= 5 and w[4] in ('in', 'out'):
 591            n[f'prefix_list_{w[4]}'] = None if negated else w[3]
 592        elif kw == 'filter-list' and len(w) >= 5 and w[4] in ('in', 'out'):
 593            n[f'filter_list_{w[4]}'] = None if negated else w[3]
 594        elif kw == 'soft-reconfiguration':
 595            n['soft_reconfiguration'] = not negated
 596        elif kw == 'send-community':
 597            n['send_community'] = not negated
 598        elif kw == 'activate':
 599            n['activate'] = not negated
 600        return
 601    if w[0] == 'bgp' and len(w) >= 2:
 602        if w[1] == 'router-id' and len(w) >= 3:
 603            rb['router_id'] = None if negated else w[2]
 604        elif w[1] == 'ebgp-requires-policy':
 605            rb['ebgp_requires_policy'] = not negated
 606        elif w[1] == 'network' and len(w) >= 3 and w[2] == 'import-check':
 607            rb['network_import_check'] = not negated
 608        elif w[1] == 'default' and len(w) >= 3 and w[2] == 'ipv4-unicast':
 609            rb['default_ipv4_unicast'] = not negated
 610        elif w[1] == 'default' and len(w) >= 4 and w[2] == 'local-preference':
 611            rb['default_local_preference'] = None if negated else int(w[3])
 612        elif w[1] == 'bestpath' and len(w) >= 3:
 613            rb['bestpath'].append(" ".join(w[2:]))
 614        elif w[1] == 'cluster-id' and len(w) >= 3:
 615            rb['cluster_id'] = None if negated else w[2]
 616        return
 617    if w[0] == 'network' and len(w) >= 2:
 618        if negated:
 619            if w[1] in rb['networks']:
 620                rb['networks'].remove(w[1])
 621        else:
 622            rb['networks'].append(w[1])
 623    elif w[0] == 'aggregate-address' and len(w) >= 2 and not negated:
 624        rb['aggregate_addresses'].append((w[1], w[2:]))
 625    elif w[0] == 'redistribute' and len(w) >= 2 and not negated:
 626        rb['redistribute'].append(w[1])
 627    elif w[0] == 'timers' and len(w) >= 4 and w[1] == 'bgp' and w[2].isdigit() and w[3].isdigit():
 628        rb['timers'] = (int(w[2]), int(w[3]))

 631def parse_frr_config(text: str) -> Dict[str, Any]:
 632    """Parse an FRR configuration (``show running-config`` or ``frr.conf``).
 633
 634    Returns::
 635
 636        {'hostname': str | None,
 637         'static_routes': [(prefix, nexthop)],       # nexthop may be 'blackhole' / 'Null0'
 638         'router_ospf': {'present': bool, 'router_id': str | None,
 639                         'networks': [(prefix, area)], 'passive_interfaces': [...],
 640                         'passive_default': bool,
 641                         'areas': {area: {'stub', 'nssa', 'no_summary', 'ranges': [prefix],
 642                                          'authentication', 'default_cost'}},
 643                         'redistribute': [protocol, ...],
 644                         'default_information_originate': bool,
 645                         'default_information_always': bool,
 646                         'auto_cost_reference_bandwidth': int | None},
 647         'interfaces': {name: {'cost', 'priority', 'hello_interval', 'dead_interval',
 648                               'authentication' ('message-digest' | 'simple' | 'null' | 'key-chain'),
 649                               'message_digest_keys': {id: key}, 'authentication_key',
 650                               'area', 'bandwidth', 'passive' (True/False/None), 'network_type',
 651                               'addresses': ['10.0.0.1/32', ...]}},
 652         'router_bgp': {'present': bool, 'as': int | None, 'router_id': str | None,
 653                        'ebgp_requires_policy': bool, 'network_import_check': bool,
 654                        'default_ipv4_unicast': bool, 'bestpath': ['compare-routerid', ...],
 655                        'cluster_id', 'default_local_preference', 'timers': (keepalive, hold) | None,
 656                        'neighbors': {peer: {'remote_as': int | str | None, 'peer_group', 'description',
 657                                             'update_source', 'timers': (keepalive, hold) | None,
 658                                             'shutdown': bool, 'password', 'ebgp_multihop': int | None,
 659                                             'next_hop_self': bool, 'route_reflector_client': bool,
 660                                             'default_originate': bool, 'default_originate_route_map',
 661                                             'route_map_in', 'route_map_out', 'prefix_list_in',
 662                                             'prefix_list_out', 'filter_list_in', 'filter_list_out',
 663                                             'soft_reconfiguration': bool, 'send_community': bool,
 664                                             'activate': bool | None}},
 665                        'networks': [prefix], 'aggregate_addresses': [(prefix, [option, ...])],
 666                        'redistribute': [protocol, ...]},
 667         'prefix_lists': {name: [{'seq': int | None, 'action': 'permit' | 'deny', 'prefix': str,
 668                                  'ge': int | None, 'le': int | None}]},
 669         'route_maps': {name: [{'seq': int, 'action': 'permit' | 'deny',
 670                                'match': ['ip address prefix-list P', ...],
 671                                'set': ['local-preference 200', ...]}]},
 672         'as_path_lists': {name: [(seq | None, action, regexp)]},
 673         'community_lists': {name: [(seq | None, action, 'community ...')]}}
 674
 675    Area ids are normalised to dotted form (``area 1`` -> ``'0.0.0.1'``).  Both the
 676    router-level ``passive-interface X`` and the interface-level ``ip ospf passive``
 677    syntaxes are recognised; ``interfaces[X]['passive']`` reflects only the
 678    interface-level statement, ``router_ospf['passive_interfaces']`` the router-level one.
 679    In ``router bgp`` the statements of the ``address-family ipv4 unicast`` block and the
 680    router-level ones are merged (FRR moves ``network``, ``next-hop-self``, ``route-map``...
 681    into the address-family block in ``show running-config``).
 682    """
 683    cfg: Dict[str, Any] = {
 684        'hostname': None,
 685        'static_routes': [],
 686        'router_ospf': {
 687            'present': False, 'router_id': None, 'networks': [], 'passive_interfaces': [],
 688            'passive_default': False, 'areas': {}, 'redistribute': [],
 689            'default_information_originate': False, 'default_information_always': False,
 690            'auto_cost_reference_bandwidth': None,
 691        },
 692        'interfaces': {},
 693        'router_bgp': _empty_router_bgp_config(),
 694        'prefix_lists': {},
 695        'route_maps': {},
 696        'as_path_lists': {},
 697        'community_lists': {},
 698    }
 699    section: Optional[str] = None  # 'router_ospf' | 'router_bgp' | 'interface' | 'route_map' | 'other' | None
 700    iface: Dict[str, Any] | None = None
 701    rm_entry: Dict[str, Any] | None = None
 702    ro = cfg['router_ospf']
 703    rb = cfg['router_bgp']
 704
 705    def neighbor_cfg(peer: str) -> Dict[str, Any]:
 706        if peer not in rb['neighbors']:
 707            rb['neighbors'][peer] = _empty_bgp_neighbor_config()
 708        return rb['neighbors'][peer]
 709
 710    def seq_of(w: List[str]):
 711        """Return (seq, rest) for a 'seq N ...' prefix of the word list."""
 712        if len(w) >= 2 and w[0] == 'seq' and w[1].isdigit():
 713            return int(w[1]), w[2:]
 714        return None, w
 715
 716    def area_cfg(area: str) -> Dict[str, Any]:
 717        key = normalize_area(area)
 718        if key not in ro['areas']:
 719            ro['areas'][key] = _empty_area_config()
 720        return ro['areas'][key]
 721
 722    for raw in text.splitlines():
 723        line = raw.rstrip()
 724        if not line.strip() or line.strip() == '!':
 725            continue
 726        stripped = line.strip()
 727        words = stripped.split()
 728
 729        if not line[0].isspace():
 730            # top-level line: opens a section or is a global statement
 731            if stripped in ('exit', 'end', 'exit-address-family'):
 732                section, iface, rm_entry = None, None, None
 733                continue
 734            if words[0] == 'interface' and len(words) >= 2:
 735                section = 'interface'
 736                iface = cfg['interfaces'].setdefault(words[1], _empty_interface_config())
 737                continue
 738            if words[0] == 'router' and len(words) >= 2 and words[1] == 'ospf':
 739                section = 'router_ospf'
 740                iface = None
 741                ro['present'] = True
 742                continue
 743            if words[0] == 'router' and len(words) >= 3 and words[1] == 'bgp':
 744                section, iface = 'router_bgp', None
 745                rb['present'] = True
 746                rb['as'] = int(words[2]) if words[2].isdigit() else words[2]
 747                continue
 748            if words[0] == 'route-map' and len(words) >= 4 and words[2] in ('permit', 'deny'):
 749                section, iface = 'route_map', None
 750                rm_entry = {'seq': int(words[3]) if words[3].isdigit() else words[3], 'action': words[2],
 751                            'match': [], 'set': []}
 752                cfg['route_maps'].setdefault(words[1], []).append(rm_entry)
 753                continue
 754            if words[0] == 'router' or words[0] in ('vrf', 'route-map', 'key', 'line', 'segment-routing'):
 755                section, iface = 'other', None
 756                continue
 757            section, iface, rm_entry = None, None, None
 758            if words[0] == 'hostname' and len(words) >= 2:
 759                cfg['hostname'] = words[1]
 760            elif words[0] == 'ip' and len(words) >= 4 and words[1] == 'route':
 761                cfg['static_routes'].append((words[2], words[3]))
 762            elif words[:2] == ['ip', 'prefix-list'] and len(words) >= 5:
 763                # ip prefix-list NAME [seq N] permit|deny PREFIX [ge X] [le Y]
 764                seq, rest = seq_of(words[3:])
 765                if len(rest) >= 2 and rest[0] in ('permit', 'deny'):
 766                    entry = {'seq': seq, 'action': rest[0], 'prefix': rest[1], 'ge': None, 'le': None}
 767                    for i in range(2, len(rest) - 1):
 768                        if rest[i] in ('ge', 'le') and rest[i + 1].isdigit():
 769                            entry[rest[i]] = int(rest[i + 1])
 770                    cfg['prefix_lists'].setdefault(words[2], []).append(entry)
 771            elif words[:3] == ['bgp', 'as-path', 'access-list'] and len(words) >= 6:
 772                # bgp as-path access-list NAME [seq N] permit|deny REGEXP
 773                seq, rest = seq_of(words[4:])
 774                if len(rest) >= 2 and rest[0] in ('permit', 'deny'):
 775                    cfg['as_path_lists'].setdefault(words[3], []).append((seq, rest[0], " ".join(rest[1:])))
 776            elif words[:2] == ['bgp', 'community-list'] and len(words) >= 5:
 777                # bgp community-list [standard|expanded] NAME [seq N] permit|deny COMMUNITY...
 778                rest = words[2:]
 779                if rest and rest[0] in ('standard', 'expanded'):
 780                    rest = rest[1:]
 781                if len(rest) >= 3:
 782                    name = rest[0]
 783                    seq, rest2 = seq_of(rest[1:])
 784                    if len(rest2) >= 2 and rest2[0] in ('permit', 'deny'):
 785                        cfg['community_lists'].setdefault(name, []).append((seq, rest2[0], " ".join(rest2[1:])))
 786            continue
 787
 788        # indented line: belongs to the current section
 789        if section == 'interface' and iface is not None:
 790            if words[:2] == ['ip', 'ospf'] and len(words) >= 3:
 791                sub = words[2]
 792                arg = words[3] if len(words) >= 4 else None
 793                if sub == 'cost' and arg:
 794                    iface['cost'] = int(arg)
 795                elif sub == 'priority' and arg:
 796                    iface['priority'] = int(arg)
 797                elif sub == 'hello-interval' and arg:
 798                    iface['hello_interval'] = int(arg)
 799                elif sub == 'dead-interval' and arg:
 800                    iface['dead_interval'] = int(arg) if arg.isdigit() else arg
 801                elif sub == 'authentication':
 802                    iface['authentication'] = arg if arg else 'simple'
 803                elif sub == 'authentication-key' and arg:
 804                    iface['authentication_key'] = arg
 805                elif sub == 'message-digest-key' and len(words) >= 6 and words[4] == 'md5':
 806                    iface['message_digest_keys'][int(arg)] = words[5]
 807                elif sub == 'area' and arg:
 808                    iface['area'] = normalize_area(arg)
 809                elif sub == 'passive':
 810                    iface['passive'] = True
 811                elif sub == 'network' and arg:
 812                    iface['network_type'] = arg
 813            elif words[:3] == ['no', 'ip', 'ospf'] and len(words) >= 4 and words[3] == 'passive':
 814                iface['passive'] = False
 815            elif words[0] == 'bandwidth' and len(words) >= 2:
 816                iface['bandwidth'] = int(words[1])
 817            elif words[:2] == ['ip', 'address'] and len(words) >= 3:
 818                iface['addresses'].append(words[2])
 819            continue
 820
 821        if section == 'route_map' and rm_entry is not None:
 822            if words[0] == 'match' and len(words) >= 2:
 823                rm_entry['match'].append(" ".join(words[1:]))
 824            elif words[0] == 'set' and len(words) >= 2:
 825                rm_entry['set'].append(" ".join(words[1:]))
 826            continue
 827
 828        if section == 'router_bgp':
 829            _parse_router_bgp_line(rb, words, neighbor_cfg)
 830            continue
 831
 832        if section == 'router_ospf':
 833            if words[0] in ('ospf', 'router-id') and 'router-id' in words:
 834                ro['router_id'] = words[-1]
 835            elif words[0] == 'network' and len(words) >= 4 and words[2] == 'area':
 836                ro['networks'].append((words[1], normalize_area(words[3])))
 837            elif words[0] == 'passive-interface' and len(words) >= 2:
 838                if words[1] == 'default':
 839                    ro['passive_default'] = True
 840                else:
 841                    ro['passive_interfaces'].append(words[1])
 842            elif words[0] == 'no' and len(words) >= 3 and words[1] == 'passive-interface':
 843                if words[2] in ro['passive_interfaces']:
 844                    ro['passive_interfaces'].remove(words[2])
 845            elif words[0] == 'area' and len(words) >= 3:
 846                a = area_cfg(words[1])
 847                kw = words[2]
 848                if kw == 'stub':
 849                    a['stub'] = True
 850                    a['no_summary'] = 'no-summary' in words[3:]
 851                elif kw == 'nssa':
 852                    a['nssa'] = True
 853                    a['no_summary'] = 'no-summary' in words[3:]
 854                elif kw == 'range' and len(words) >= 4:
 855                    a['ranges'].append(words[3])
 856                elif kw == 'authentication':
 857                    a['authentication'] = words[3] if len(words) >= 4 else 'simple'
 858                elif kw == 'default-cost' and len(words) >= 4:
 859                    a['default_cost'] = int(words[3])
 860            elif words[0] == 'redistribute' and len(words) >= 2:
 861                ro['redistribute'].append(words[1])
 862            elif words[:2] == ['default-information', 'originate']:
 863                ro['default_information_originate'] = True
 864                ro['default_information_always'] = 'always' in words[2:]
 865            elif words[:2] == ['auto-cost', 'reference-bandwidth'] and len(words) >= 3:
 866                ro['auto_cost_reference_bandwidth'] = int(words[2])
 867            continue
 868
 869    return cfg

 881def expected_auto_cost(reference_bandwidth: int, interface_bandwidth: int) -> int:
 882    """OSPF cost FRR computes for an interface: ``round(reference / bandwidth)`` clamped to
 883    ``[1, 65535]`` (both in Mbit/s; the default reference is 100000 and a veth reports 10000,
 884    hence the default cost of 10)."""

 886        return 10

 888    return max(1, min(65535, cost))

 929
 930def parse_bgp_summary(data: Any) -> Dict[str, Any]:
 931    """Parse the JSON of ``show bgp ipv4 unicast summary``.
 932
 933    Accepts the flat address-family object or the ``{'ipv4Unicast': {...}}`` wrapper of
 934    ``show bgp summary json``.  Returns ``{}`` when no BGP process answers, else::
 935
 936        {'router_id': '10.0.0.1', 'as': 65001,
 937         'peers': {'10.1.1.2': {'remote_as': 65100, 'state': 'Established', 'established': True,
 938                                'policy_blocked': False,   # eBGP peer without in/out policy (RFC 8212)
 939                                'pfx_rcd': 3, 'pfx_snt': 1, 'description': 'ISP-A',
 940                                'uptime': '00:12:34'}}}
 941    """
 942    if not isinstance(data, dict):
 943        return {}
 944    if 'ipv4Unicast' in data and isinstance(data['ipv4Unicast'], dict):
 945        data = data['ipv4Unicast']
 946    if 'routerId' not in data and 'peers' not in data:
 947        return {}

382    """Parse the text of ``show ip ospf route``.
383
384    Network entries (``N``) and discard entries (``D``, created by ``area range``) are
385    keyed by prefix, router entries (``R``) by router id::
386
387        {'10.0.5.0/24': {'kind': 'N', 'type': 'IA', 'cost': 30, 'ext_cost': None, 'tag': None,
388                         'area': '0.0.0.0', 'abr': False, 'asbr': False,

392
393    ``type`` is ``'intra'`` for plain intra-area entries, ``'IA'``, ``'E1'``, ``'E2'``,
394    ``'N1'`` or ``'N2'``.  ``nexthops`` holds ``(ip, interface)`` pairs, ``ip`` being
395    ``None`` for ``directly attached to`` entries.
396    """
397    result: Dict[str, Dict[str, Any]] = {}
398    current: Dict[str, Any] | None = None
399
400    for line in output.splitlines():
401        m = _OSPF_ROUTE_RE.match(line.strip())
402        if m:
403            rest = m.group('rest') or ''
404            area_m = re.search(r'area:\s*(\S+?)(?:,|\s|$)', rest)
405            tag_m = re.search(r'tag:\s*(\d+)', rest)
406            current = {
407                'kind': m.group('kind'),
408                'type': m.group('type') or m.group('type2') or 'intra',
409                'cost': int(m.group('cost')) if m.group('cost') else None,
410                'ext_cost': int(m.group('ext_cost')) if m.group('ext_cost') else None,
411                'tag': int(tag_m.group(1)) if tag_m else None,
412                'area': area_m.group(1) if area_m else None,
413                'abr': 'ABR' in rest,
414                'asbr': 'ASBR' in rest,
415                'discard': bool(m.group('discard')),
416                'nexthops': [],
417            }
418            result[m.group('dest')] = current
419            continue
420        if current is None:
421            continue
422        m = re.search(r'via (\S+?),\s*(\S+)', line)
423        if m:
424            current['nexthops'].append((m.group(1), m.group(2)))
425            continue
426        m = re.search(r'directly attached to (\S+)', line)
427        if m:
428            current['nexthops'].append((None, m.group(1)))
429            continue
430
431    return result
