# configuration file /etc/nginx/conf.d/cdn.conf:
server {
    listen 443 quic reuseport;
    listen 443 ssl;
    http2 on;
    http3 on;
    server_name cdn.quic.tp;
    ssl_certificate /etc/nginx/ssl/cdn.quic.tp.crt;
    ssl_certificate_key /etc/nginx/ssl/cdn.quic.tp.key;
    add_header Alt-Svc 'h3=":443"; ma=86400' always;
    root /var/www/cdn;
    location = /big.bin {
        limit_rate 1m;
    }
    location = /order {
        default_type text/plain;
        return 200 "order accepted\n";
    }
}
