# configuration file /etc/nginx/conf.d/shop.conf:
server_tokens off;

upstream shop_app {
    server web1:8080 weight=2;
    server web2:8080;
    server web3:8080 backup;
    keepalive 16;
}

upstream shop_api {
    server web2:8081;
    server web3:8081;
}

limit_req_zone $binary_remote_addr zone=api:10m rate=5r/s;

# configuration file /etc/nginx/sites-enabled/shop:
# unknown names: nothing is served
server {
    listen 80 default_server;
    listen 443 ssl default_server;
    ssl_reject_handshake on;
    server_name _;
    return 444;
}

# www.shop.tp in clear: everything goes to HTTPS
server {
    listen 80;
    server_name www.shop.tp;
    return 301 https://$host$request_uri;
}

# the former name of the site
server {
    listen 80;
    server_name old.shop.tp;
    return 301 https://www.shop.tp$request_uri;
}

# the API under its own name
server {
    listen 80;
    server_name api.shop.tp;
    include /etc/nginx/snippets/shop-proxy.conf;
    location / {
        limit_req zone=api burst=5 nodelay;
        limit_req_status 429;
        proxy_pass http://shop_api;
    }
}

# the shop
server {
    listen 443 ssl http2;
    server_name www.shop.tp;
    ssl_certificate     /etc/ssl/shop/www.shop.tp.crt;
    ssl_certificate_key /etc/ssl/shop/www.shop.tp.key;
    add_header Strict-Transport-Security "max-age=31536000" always;
    include /etc/nginx/snippets/shop-proxy.conf;
    gzip_types text/plain application/json;
    proxy_read_timeout 3s;
    error_page 502 503 504 /maintenance.html;

    location / {
        proxy_pass http://shop_app;
    }
    location /api/ {
        limit_req zone=api burst=5 nodelay;
        limit_req_status 429;
        proxy_pass http://shop_api;
    }
    location /static/ {
        root /var/www/shop;
        expires 1h;
    }
    location /legacy/ {
        proxy_pass http://shop_app/;
    }
    location /admin/ {
        auth_basic "Shop administration";
        auth_basic_user_file /etc/nginx/htpasswd;
        proxy_pass http://shop_app;
    }
    location = /nginx_status {
        stub_status;
        allow 192.168.227.77;
        deny all;
    }
    location /reports/ {
        proxy_pass http://web3:8090;
    }
    location = /maintenance.html {
        root /var/www/shop;
        internal;
    }
    location /catalog/ {
        proxy_pass http://127.0.0.1:6081;
    }
}
